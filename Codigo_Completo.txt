import sys
import os
from pathlib import Path

def obter_diretorio_base() -> Path:
    """
    Retorna o diretório permanente onde o .exe está localizado.
    Em desenvolvimento, retorna a raiz do repositório.
    """
    if getattr(sys, 'frozen', False):
        return Path(sys.executable).resolve().parent
    return Path(__file__).resolve().parent.parent

def obter_diretorio_recursos() -> Path:
    """
    Retorna o diretório dos arquivos estáticos do frontend com proteção
    para todas as versões do PyInstaller (_MEIPASS, _internal ou raiz).
    """
    if getattr(sys, 'frozen', False):
        # 1. Verifica no _MEIPASS temporário
        if hasattr(sys, '_MEIPASS'):
            p_mei = Path(sys._MEIPASS)
            if (p_mei / "frontend" / "dist").exists():
                return p_mei
        
        # 2. Verifica dentro da pasta _internal ao lado do .exe (PyInstaller 6+ onedir)
        exe_dir = Path(sys.executable).resolve().parent
        if (exe_dir / "_internal" / "frontend" / "dist").exists():
            return exe_dir / "_internal"
            
        # 3. Verifica diretamente ao lado do .exe
        if (exe_dir / "frontend" / "dist").exists():
            return exe_dir

    return Path(__file__).resolve().parent.parent

import sqlite3
import calendar
from datetime import datetime
from contextlib import contextmanager
from backend.config import CAMINHO_BANCO

@contextmanager
def obter_conexao():
    """
    Gerenciador de contexto que garante conexão aberta com WAL ativado,
    commit automático ao sair com sucesso, rollback em exceção e fechamento garantido.
    """
    conexao = sqlite3.connect(str(CAMINHO_BANCO), timeout=30.0)
    conexao.row_factory = sqlite3.Row
    conexao.execute("PRAGMA journal_mode=WAL;")
    conexao.execute("PRAGMA foreign_keys=ON;")
    try:
        yield conexao
        conexao.commit()
    except Exception:
        conexao.rollback()
        raise
    finally:
        conexao.close()

def criar_tabelas():
    with obter_conexao() as conexao:
        cursor = conexao.cursor()
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS boletos (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                distribuidora TEXT NOT NULL,
                numero_nota TEXT,
                valor_parcela REAL NOT NULL,
                data_emissao DATE,
                data_vencimento DATE NOT NULL,
                parcela_atual INTEGER,
                total_parcelas INTEGER,
                status TEXT DEFAULT 'Aberto'
            )
        ''')
    print("[Banco] Tabela 'boletos' verificada/criada com sucesso.")

def calcular_novo_mes(data_base: datetime, meses_a_adicionar: int) -> str:
    """
    Avança exatamente N meses mantendo o dia fixo do vencimento, 
    respeitando meses de 28, 30 e 31 dias (inclusive anos bissextos).
    """
    ano = data_base.year + (data_base.month + meses_a_adicionar - 1) // 12
    mes = (data_base.month + meses_a_adicionar - 1) % 12 + 1
    ultimo_dia_mes = calendar.monthrange(ano, mes)[1]
    dia = min(data_base.day, ultimo_dia_mes)
    return datetime(ano, mes, dia).strftime('%Y-%m-%d')

def cadastrar_boleto(distribuidora, numero_nota, valor, data_emissao, data_vencimento, parcela_atual, total_parcelas):
    vencimento_base = datetime.strptime(data_vencimento, '%Y-%m-%d')
    nota_limpa = (numero_nota or 'S/N').strip()
    dist_limpa = distribuidora.strip()
    emissao_limpa = data_emissao.strip() if data_emissao else datetime.now().strftime('%Y-%m-%d')

    if parcela_atual > total_parcelas:
        raise ValueError("A parcela atual não pode ser maior que o total de parcelas.")

    with obter_conexao() as conexao:
        cursor = conexao.cursor()
        for i in range(parcela_atual, total_parcelas + 1):
            salto_meses = i - parcela_atual
            novo_vencimento = calcular_novo_mes(vencimento_base, salto_meses)
            
            cursor.execute('''
                INSERT INTO boletos (
                    distribuidora, numero_nota, valor_parcela, data_emissao, 
                    data_vencimento, parcela_atual, total_parcelas, status
                ) VALUES (?, ?, ?, ?, ?, ?, ?, 'Aberto')
            ''', (dist_limpa, nota_limpa, float(valor), emissao_limpa, novo_vencimento, i, total_parcelas))
            
    print(f"[Banco] Lote {nota_limpa} ({dist_limpa}) cadastrado: {total_parcelas - parcela_atual + 1} parcelas salvas.")

def listar_boletos():
    with obter_conexao() as conexao:
        cursor = conexao.cursor()
        cursor.execute("SELECT id, distribuidora, numero_nota, valor_parcela, data_emissao, data_vencimento, parcela_atual, total_parcelas, status FROM boletos ORDER BY data_vencimento ASC")
        return [dict(linha) for linha in cursor.fetchall()]

def formatar_data_br(data_iso: str) -> str:
    if not data_iso or len(data_iso) != 10:
        return data_iso or ""
    p = data_iso.split('-')
    return f"{p[2]}/{p[1]}/{p[0]}"

@app.get("/boletos")
def obter_boletos(
    distribuidora: Optional[str] = Query(None),
    dias_vencimento: Optional[int] = Query(None),
    status: Optional[str] = Query("Aberto")
):
    boletos_raw = db.listar_boletos()
    boletos_ativos = [b for b in boletos_raw if b["status"] != 'Excluido']

    # Mapa para consolidar o resumo completo de cada compra
    mapa_compras = {}
    for b in boletos_ativos:
        dist = (b["distribuidora"] or '').strip().upper()
        nota = (b["numero_nota"] or 'S/N').strip().upper()
        emissao = b["data_emissao"] or b["data_vencimento"]
        venc = b["data_vencimento"]
        valor = float(b["valor_parcela"])
        st = b["status"]
        chave = f"{dist}__{nota}__{b['data_emissao'] or 'SEM_EMISSAO'}"

        if chave not in mapa_compras:
            mapa_compras[chave] = {
                "inicio": emissao,
                "termino": venc,
                "total_parcelas": b["total_parcelas"],
                "valor_total": 0.0,
                "total_pago": 0.0,
                "total_pendente": 0.0,
                "qtd_pagas": 0,
                "todos_ids": [],
                "todas_parcelas_info": []
            }
        
        info = mapa_compras[chave]
        info["todos_ids"].append(b["id"])
        info["todas_parcelas_info"].append({"id": b["id"], "status_anterior": b["status"]})

        if venc > info["termino"]:
            info["termino"] = venc
        if emissao < info["inicio"]:
            info["inicio"] = emissao
            
        info["valor_total"] += valor
        if st == 'Pago':
            info["total_pago"] += valor
            info["qtd_pagas"] += 1
        else:
            info["total_pendente"] += valor

    boletos_formatados = []
    hoje = datetime.now().date()
    data_limite = hoje + timedelta(days=dias_vencimento) if dias_vencimento is not None else None

    for b in boletos_ativos:
        dist = (b["distribuidora"] or '').strip().upper()
        nota = (b["numero_nota"] or 'S/N').strip().upper()
        chave = f"{dist}__{nota}__{b['data_emissao'] or 'SEM_EMISSAO'}"
        dados_compra = mapa_compras.get(chave)

        item = {
            "id": b["id"],
            "distribuidora": b["distribuidora"],
            "numero_nota": b["numero_nota"] or 'S/N',
            "valor_parcela": b["valor_parcela"],
            "data_emissao": b["data_emissao"],
            "data_vencimento": b["data_vencimento"],
            "parcela_atual": b["parcela_atual"],
            "total_parcelas": b["total_parcelas"],
            "status": b["status"],
            "compra_chave": chave,
            "compra_resumo": {
                "data_inicio": dados_compra["inicio"] if dados_compra else b["data_vencimento"],
                "data_termino": dados_compra["termino"] if dados_compra else b["data_vencimento"],
                "total_parcelas": dados_compra["total_parcelas"] if dados_compra else b["total_parcelas"],
                "qtd_pagas": dados_compra["qtd_pagas"] if dados_compra else 0,
                "valor_total": round(dados_compra["valor_total"], 2) if dados_compra else round(b["valor_parcela"] * b["total_parcelas"], 2),
                "total_pago": round(dados_compra["total_pago"], 2) if dados_compra else 0.0,
                "total_pendente": round(dados_compra["total_pendente"], 2) if dados_compra else round(b["valor_parcela"], 2),
                "todos_ids_compra": dados_compra["todos_ids"] if dados_compra else [b["id"]],
                "todas_parcelas_info": dados_compra["todas_parcelas_info"] if dados_compra else [{"id": b["id"], "status_anterior": b["status"]}]
            }
        }
        
        if status and status.lower() != 'todos':
            if item["status"].lower() != status.lower():
                continue

        if distribuidora and distribuidora.strip():
            if distribuidora.strip().lower() not in item["distribuidora"].lower():
                continue

        if dias_vencimento is not None:
            try:
                data_venc = datetime.strptime(item["data_vencimento"], '%Y-%m-%d').date()
                if not (data_venc <= data_limite):
                    continue
            except ValueError:
                continue

        boletos_formatados.append(item)
        
    return boletos_formatados

@app.post("/boletos")
def criar_boleto(novo_boleto: BoletoCreate):
    if novo_boleto.parcela_atual > novo_boleto.total_parcelas:
        raise HTTPException(status_code=400, detail="A parcela inicial não pode ser superior ao total de parcelas.")

    db.cadastrar_boleto(
        novo_boleto.distribuidora, 
        novo_boleto.numero_nota, 
        novo_boleto.valor_parcela, 
        novo_boleto.data_emissao, 
        novo_boleto.data_vencimento, 
        novo_boleto.parcela_atual, 
        novo_boleto.total_parcelas
    )
    return {"mensagem": "Boleto e parcelas projetadas salvos com sucesso!"}

@app.put("/boletos/{boleto_id}/pagar")
def marcar_como_pago(boleto_id: int):
    with db.obter_conexao() as conexao:
        cursor = conexao.cursor()
        cursor.execute("UPDATE boletos SET status = 'Pago' WHERE id = ?", (boleto_id,))
        if cursor.rowcount == 0:
            raise HTTPException(status_code=404, detail="Boleto não encontrado.")
    return {"mensagem": f"Boleto ID {boleto_id} marcado como Pago!"}

@app.post("/compras/excluir")
def excluir_compra(dados: ExcluirLotePayload):
    if not dados.ids:
        raise HTTPException(status_code=400, detail="Nenhum boleto informado para exclusão.")
    
    placeholders = ','.join(['?'] * len(dados.ids))
    with db.obter_conexao() as conexao:
        cursor = conexao.cursor()
        cursor.execute(f"UPDATE boletos SET status = 'Excluido' WHERE id IN ({placeholders})", dados.ids)
    return {"mensagem": "Compra excluída com sucesso!"}

@app.post("/compras/restaurar")
def restaurar_compra(dados: RestaurarLotePayload):
    if not dados.parcelas:
        raise HTTPException(status_code=400, detail="Nenhuma parcela informada para restauração.")

    with db.obter_conexao() as conexao:
        cursor = conexao.cursor()
        for p in dados.parcelas:
            cursor.execute("UPDATE boletos SET status = ? WHERE id = ?", (p.status_anterior, p.id))
    return {"mensagem": "Compra restaurada com sucesso!"}

import sqlite3
from datetime import datetime
from backend.config import CAMINHO_BANCO, PASTA_BACKUPS

def fazer_backup_banco():
    """
    Realiza backup a quente e consistente utilizando a API oficial de backup do SQLite,
    evitando corrupção de páginas e compatível com modo WAL.
    """
    if not CAMINHO_BANCO.exists():
        return

    PASTA_BACKUPS.mkdir(parents=True, exist_ok=True)

    data_hoje = datetime.now().strftime('%Y%m%d')
    nome_backup = f"oficina_{data_hoje}.db.bak"
    caminho_destino = PASTA_BACKUPS / nome_backup

    if caminho_destino.exists():
        try:
            mtime_banco = CAMINHO_BANCO.stat().st_mtime
            mtime_backup = caminho_destino.stat().st_mtime
            if mtime_banco <= mtime_backup:
                return
        except Exception:
            pass

    try:
        # Backup online atômico e transacional nativo do SQLite
        with sqlite3.connect(str(CAMINHO_BANCO)) as src:
            with sqlite3.connect(str(caminho_destino)) as dst:
                src.backup(dst)
        
        print(f"[Segurança] Backup diário registrado com sucesso: {nome_backup}")

        # Retenção: Mantém os 20 backups mais recentes
        backups = sorted(PASTA_BACKUPS.glob("oficina_*.db.bak"), key=lambda p: p.stat().st_mtime)
        while len(backups) > 20:
            antigo = backups.pop(0)
            try:
                antigo.unlink()
            except Exception:
                pass
    except Exception as e:
        print(f"[Erro de Segurança] Falha ao criar backup: {e}")

  if (!dataIso) return ''

  if (partes.length !== 3) return dataIso
  return `${partes[2]}/${partes[1]}/${partes[0]}`

const apagarCompra = async (compra) => {
  const confirmou = confirm(
    `Tem certeza que deseja apagar a compra "${compra.distribuidora}" (NF: ${compra.numero_nota}) com todas as ${compra.total_parcelas} parcelas registradas?`
  )
  if (!confirmou) return

  try {
    // Captura os IDs e status de TODAS as parcelas reais da compra
    const idsParaExcluir = compra.todos_ids
    const parcelasBackup = compra.todas_parcelas_info

    const res = await fetch(obterApiUrl('/compras/excluir'), {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ids: idsParaExcluir })

    if (toastUndo.value.intervaloId) clearInterval(toastUndo.value.intervaloId)

      toastUndo.value.tempoRestante--
      if (toastUndo.value.tempoRestante <= 0) {

      }

  } catch (erro) {
    alert("Erro ao excluir: " + erro.message)
  }
}

}

  const grupos = {}

  boletosGestao.value.forEach(b => {
    const chave = b.compra_chave
    const resumo = b.compra_resumo

    if (!grupos[chave]) {
      grupos[chave] = {
        chave,
        distribuidora: b.distribuidora,
        numero_nota: b.numero_nota,
        data_inicio: resumo.data_inicio,
        data_termino: resumo.data_termino,

        todos_ids: resumo.todos_ids_compra,
        todas_parcelas_info: resumo.todas_parcelas_info,
        parcelas: []
      }
    }
    grupos[chave].parcelas.push(b)
  })

  // Ordena parcelas internas numericamente
  Object.values(grupos).forEach(grupo => {
    grupo.parcelas.sort((a, b) => a.parcela_atual - b.parcela_atual)
  })

})

<script setup>
import { ref, computed, watch } from 'vue'
import { Doughnut, Bar, Line } from 'vue-chartjs'
import { 
  Chart as ChartJS, Title, Tooltip, Legend, ArcElement, 
  BarElement, CategoryScale, LinearScale, PointElement, LineElement, Filler 
} from 'chart.js'

ChartJS.register(
  Title, Tooltip, Legend, ArcElement, 
  BarElement, CategoryScale, LinearScale, PointElement, LineElement, Filler
)

const props = defineProps({
  boletos: {
    type: Array,
    required: true
  }
})

const tipoGrafico = ref('donut')

const paletaCores = [
  '#00d2ff', '#10b981', '#6366f1', '#f59e0b', '#ec4899',
  '#06b6d4', '#14b8a6', '#8b5cf6', '#3b82f6', '#f43f5e'
]

const gerarCor = (index) => {
  if (index < paletaCores.length) return paletaCores[index]
  const hue = (index * 137.5) % 360
  return `hsl(${Math.round(hue)}, 80%, 55%)`
}

const formatarMes = (anoMes) => {
  if (!anoMes) return ''
  const partes = anoMes.split('-')
  if (partes.length < 2) return anoMes
  const mesesNomes = ['Jan', 'Fev', 'Mar', 'Abr', 'Mai', 'Jun', 'Jul', 'Ago', 'Set', 'Out', 'Nov', 'Dez']
  const mesIndex = parseInt(partes[1], 10) - 1
  return `${mesesNomes[mesIndex] || partes[1]} / ${partes[0]}`
}

const formatarMoeda = (val) => {
  return Number(val || 0).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' })
}

// Meses com contas em aberto
const mesesDisponiveis = computed(() => {
  const meses = new Set()
  props.boletos.forEach(b => {
    if (b.data_vencimento && b.status !== 'Pago') {
      meses.add(b.data_vencimento.substring(0, 7))
    }
  })
  return Array.from(meses).sort()
})

const mesSelecionado = ref('')

// Prioriza o mês atual; se não houver pendências nele, pega o primeiro com pendências
watch(mesesDisponiveis, (novosMeses) => {
  if (novosMeses.length > 0) {
    const mesAtualIso = new Date().toISOString().substring(0, 7)
    if (novosMeses.includes(mesAtualIso)) {
      mesSelecionado.value = mesAtualIso
    } else if (!mesSelecionado.value || !novosMeses.includes(mesSelecionado.value)) {
      mesSelecionado.value = novosMeses[0]
    }
  } else {
    mesSelecionado.value = ''
  }
}, { immediate: true })

const analiseFornecedoresMes = computed(() => {
  const totais = {}
  let totalGeralMes = 0

  props.boletos.forEach(b => {
    const mesAno = b.data_vencimento ? b.data_vencimento.substring(0, 7) : ''
    if (mesAno === mesSelecionado.value && b.status !== 'Pago') {
      const dist = b.distribuidora || 'Outros'
      totais[dist] = (totais[dist] || 0) + b.valor_parcela
      totalGeralMes += b.valor_parcela
    }
  })

  const lista = Object.keys(totais).map((dist, idx) => ({
    distribuidora: dist,
    valor: totais[dist],
    porcentagem: totalGeralMes > 0 ? ((totais[dist] / totalGeralMes) * 100).toFixed(1) : '0.0',
    cor: gerarCor(idx)
  })).sort((a, b) => b.valor - a.valor)

  return { lista, totalGeralMes }
})

const dadosDonut = computed(() => {
  const { lista } = analiseFornecedoresMes.value
  return {
    labels: lista.map(i => i.distribuidora),
    datasets: [{
      backgroundColor: lista.map(i => i.cor),
      data: lista.map(i => i.valor),
      borderWidth: 2,
      borderColor: '#0f172a',
      hoverOffset: 8
    }]
  }
})

const dadosBarras = computed(() => {
  const { lista } = analiseFornecedoresMes.value
  return {
    labels: lista.map(i => i.distribuidora),
    datasets: [{
      label: 'Valor a Pagar',
      backgroundColor: lista.map(i => i.cor),
      borderRadius: 6,
      data: lista.map(i => i.valor)
    }]
  }
})

const dadosLinhaDoTempo = computed(() => {
  const totaisPorMes = {}

  props.boletos.forEach(b => {
    if (b.data_vencimento && b.status !== 'Pago') {
      const mesAno = b.data_vencimento.substring(0, 7)
      totaisPorMes[mesAno] = (totaisPorMes[mesAno] || 0) + b.valor_parcela
    }
  })

  const mesesOrdenados = Object.keys(totaisPorMes).sort()
  return {
    labels: mesesOrdenados.map(formatarMes),
    datasets: [{
      label: 'Total de Contas a Pagar',
      borderColor: '#00d2ff',
      backgroundColor: 'rgba(0, 210, 255, 0.12)',
      borderWidth: 3,
      fill: true,
      tension: 0.35,
      pointBackgroundColor: '#10b981',
      pointBorderColor: '#ffffff',
      pointRadius: 5,
      pointHoverRadius: 7,
      data: mesesOrdenados.map(m => totaisPorMes[m])
    }]
  }
})

const opcoesDonut = {
  responsive: true,
  maintainAspectRatio: false,
  cutout: '74%',
  plugins: {
    legend: { display: false },
    tooltip: {
      backgroundColor: '#0f172a',
      borderColor: '#334155',
      borderWidth: 1,
      padding: 12,
      callbacks: { label: (c) => ` ${formatarMoeda(c.parsed)}` }
    }
  }
}

const opcoesGeraisCartesianas = {
  responsive: true,
  maintainAspectRatio: false,
  plugins: {
    legend: { display: false },
    tooltip: {
      backgroundColor: '#0f172a',
      borderColor: '#334155',
      borderWidth: 1,
      padding: 12,
      callbacks: { label: (c) => ` ${formatarMoeda(c.parsed.y !== undefined ? c.parsed.y : c.parsed)}` }
    }
  },
  scales: {
    x: {
      ticks: { color: '#94a3b8', font: { size: 11 } },
      grid: { color: '#1e293b' }
    },
    y: {
      ticks: { 
        color: '#94a3b8', 
        callback: (val) => 'R$ ' + Number(val).toLocaleString('pt-BR') 
      },
      grid: { color: '#1e293b' }
    }
  }
}
</script>
